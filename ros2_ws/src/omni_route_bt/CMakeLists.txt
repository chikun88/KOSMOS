cmake_minimum_required(VERSION 3.8)
project(omni_route_bt)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(nav2_behavior_tree REQUIRED)
find_package(nav2_util REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(rclcpp REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(yaml_cpp_vendor REQUIRED)
find_package(yaml-cpp REQUIRED)
add_library(omni_remove_passed_bucket_goals_bt_node SHARED src/remove_passed_bucket_goals.cpp)
ament_target_dependencies(omni_remove_passed_bucket_goals_bt_node
  nav2_behavior_tree nav2_util geometry_msgs rclcpp tf2_ros)
target_link_libraries(omni_remove_passed_bucket_goals_bt_node yaml-cpp)
target_compile_definitions(omni_remove_passed_bucket_goals_bt_node PRIVATE BT_PLUGIN_EXPORT)
install(TARGETS omni_remove_passed_bucket_goals_bt_node LIBRARY DESTINATION lib)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_bucket_passage test/test_bucket_passage.cpp)
  target_link_libraries(test_bucket_passage omni_remove_passed_bucket_goals_bt_node)
  target_compile_definitions(test_bucket_passage PRIVATE
    TEST_PLUGIN_PATH="$<TARGET_FILE:omni_remove_passed_bucket_goals_bt_node>")
  ament_target_dependencies(test_bucket_passage nav2_behavior_tree nav2_util rclcpp tf2_ros)
endif()
ament_package()
